function switchToWeek(newIndex, direction) { if (!timetableData || !timetableData.weeks || timetableData.weeks.length === 0) return; if (newIndex < 0 || newIndex >= timetableData.weeks.length) return; // 推挤动画 const wrapper = document.getElementById('timetableWrapper'); const offset = direction === 'left' ? -20 : 20; // 左滑->右推？根据滑动方向：左滑（下一周）则新内容从右侧进入，旧内容向左移出？我们简单做左右位移 // 先设置transform wrapper.style.transition = 'transform 0.15s ease'; wrapper.style.transform = `translateX(${offset}px)`; setTimeout(() => { // 切换 currentWeekIndex = newIndex; weekSelect.value = newIndex; renderWeekTimetable(timetableData, newIndex); // 重置transform wrapper.style.transition = 'none'; wrapper.style.transform = 'translateX(0)'; // 强制回流后恢复过渡 void wrapper.offsetHeight; wrapper.style.transition = 'transform 0.15s ease'; // 可选：再做一个轻微回弹 setTimeout(() => { wrapper.style.transform = 'translateX(0)'; }, 10); }, 150); // 但这样渲染会覆盖，需要等待。 }
